local function new(cls, props, parent)
	local i = Instance.new(cls)
	for k, v in pairs(props or {}) do i[k] = v end
	i.Parent = parent
	return i
end

local function hiddenRoot()
	for _, fn in ipairs({ gethui, get_hidden_gui, gethiddengui }) do
		if type(fn) == "function" then
			local ok, r = pcall(fn)
			if ok and r then return r end
		end
	end
	return Players.LocalPlayer:WaitForChild("PlayerGui")
end

local function getChildren(obj)
	if obj.ClassName == "CoreGui" and type(gethui) == "function" then
		local ok, r = pcall(gethui)
		if ok and r then
			local ok2, ch = pcall(function() return r:GetChildren() end)
			if ok2 and ch then return ch end
		end
	end
	local ok, res = pcall(function() return obj:GetChildren() end)
	return (ok and res) or {}
end

--[[ ---------- FORMATAÇÃO / EDIÇÃO DE VALORES ---------- ]]
local function fmt(v)
	local t = typeof(v)
	if t == "string" then return '"' .. v .. '"'
	elseif t == "boolean" or t == "number" then return tostring(v)
	elseif t == "EnumItem" then return "Enum." .. v.EnumType.Name .. "." .. v.Name
	elseif t == "Color3" then return string.format("%d, %d, %d", v.R*255, v.G*255, v.B*255)
	elseif t == "Vector3" then return string.format("%.1f, %.1f, %.1f", v.X, v.Y, v.Z)
	elseif t == "Vector2" then return string.format("%.1f, %.1f", v.X, v.Y)
	elseif t == "UDim2" then return string.format("{%g,%g},{%g,%g}", v.X.Scale, v.X.Offset, v.Y.Scale, v.Y.Offset)
	elseif t == "BrickColor" then return v.Name
	elseif t == "Instance" then
		local ok, s = pcall(function() return v:GetFullName() end)
		return ok and s or v.Name
	elseif t == "nil" then return "nil"
	elseif t == "table" then return "table"
	elseif t == "function" then return "function"
	else return t end
end

local function editText(v)
	local t = typeof(v)
	if t == "string" then return v
	elseif t == "number" then return tostring(v)
	elseif t == "Color3" then return string.format("%d, %d, %d", v.R*255, v.G*255, v.B*255)
	elseif t == "Vector3" then return string.format("%.2f, %.2f, %.2f", v.X, v.Y, v.Z)
	elseif t == "UDim2" then return string.format("{%g, %g}, {%g, %g}", v.X.Scale, v.X.Offset, v.Y.Scale, v.Y.Offset)
	elseif t == "EnumItem" then return "Enum." .. v.EnumType.Name .. "." .. v.Name
	elseif t == "BrickColor" then return v.Name
	else return fmt(v) end
end

local function parseEdit(text, current)
	local t = typeof(current)
	if t == "string" then return text
	elseif t == "number" then return tonumber(text)
	elseif t == "Color3" then
		local r, g, b = text:match("([%d%.]+)%s*,%s*([%d%.]+)%s*,%s*([%d%.]+)")
		if r then return Color3.fromRGB(tonumber(r), tonumber(g), tonumber(b)) end
	elseif t == "Vector3" then
		local x, y, z = text:match("([%-eE%d%.]+)%s*,%s*([%-eE%d%.]+)%s*,%s*([%-eE%d%.]+)")
		if x then return Vector3.new(tonumber(x), tonumber(y), tonumber(z)) end
	elseif t == "UDim2" then
		local a, b, c, d = text:match("{([%-eE%d%.]+)%s*,%s*([%-eE%d%.]+)}%s*,%s*{([%-eE%d%.]+)%s*,%s*([%-eE%d%.]+)}")
		if a then return UDim2.new(tonumber(a), tonumber(b), tonumber(c), tonumber(d)) end
	elseif t == "EnumItem" then
		local et, ei = text:match("^Enum%.(%w+)%.(%w+)$")
		if et then
			local ok, res = pcall(function() return Enum[et][ei] end)
			if ok then return res end
		end
	elseif t == "BrickColor" then
		local ok, res = pcall(function() return BrickColor.new(text) end)
		if ok then return res end
	end
	return nil
end

local ROW_H = 28
local function clearList(scroll)
	for _, c in ipairs(scroll:GetChildren()) do if c:IsA("GuiObject") then c:Destroy() end end
end

local function makeRow(obj, depth)
	local hasKids = #getChildren(obj) > 0
	local row = new("TextButton", { Size = UDim2.new(1, 0, 0, ROW_H), AutoButtonColor = false,
		BackgroundColor3 = THEME.sel, BackgroundTransparency = 1, BorderSizePixel = 0, ClipsDescendants = true, Text = "" })
	rowRefs[obj] = row
	local pad = depth * 14

	if hasKids then
		local arrow = new("TextButton", { Size = UDim2.new(0, 16, 1, 0), Position = UDim2.new(0, pad + 2, 0, 0),
			BackgroundTransparency = 1, Text = "▶", TextSize = 10, TextColor3 = THEME.dim,
			Rotation = expanded[obj] and 90 or 0, Parent = row })
		arrow.MouseButton1Click:Connect(function() expanded[obj] = not expanded[obj]; renderTree() end)
	end

	new("TextLabel", { Size = UDim2.new(0, 22, 1, 0), Position = UDim2.new(0, pad + 20, 0, 0),
		BackgroundTransparency = 1, Text = iconFor(obj), TextSize = 14, Parent = row })
	new("TextLabel", { Size = UDim2.new(1, -(pad + 46), 1, 0), Position = UDim2.new(0, pad + 44, 0, 0),
		BackgroundTransparency = 1, Text = obj.Name, TextColor3 = THEME.text, TextSize = 15,
		Font = Enum.Font.SourceSans, TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd, Parent = row })

	row.MouseButton1Click:Connect(function() selectInstance(obj) end)
	row.MouseButton2Click:Connect(function() showMenu(obj, row.AbsolutePosition + Vector2.new(30, ROW_H)) end)
	bindLongPress(row, function(pos) showMenu(obj, pos) end)
	row.MouseEnter:Connect(function() if selected ~= obj then row.BackgroundTransparency = 0.5 end end)
	row.MouseLeave:Connect(function() if selected ~= obj then row.BackgroundTransparency = 1 end end)
	return row
end

renderTree = function()
	local keep = treeScroll.CanvasPosition.Y
	clearList(treeScroll)
	rowRefs = {}
	local count = 0
	local function add(o, d)
		count = count + 1
		if count > 1200 then return end
		local row = makeRow(o, d)
		row.Parent = treeScroll
		if expanded[o] then
			for _, ch in ipairs(getChildren(o)) do add(ch, d + 1) end
		end
	end
	for _, svc in ipairs(getChildren(game)) do add(svc, 0) end
	treeScroll.CanvasPosition = Vector2.new(0, keep)
	if selected and rowRefs[selected] then rowRefs[selected].BackgroundTransparency = 0 end
end

local function renderSearch(q)
	clearList(treeScroll)
	rowRefs = {}
	q = q:lower()
	local found, iter = 0, 0
	local function walk(o)
		for _, ch in ipairs(getChildren(o)) do
			iter = iter + 1
			if iter % 400 == 0 then task.wait() end
			if found >= 150 then return end
			if ch.Name:lower():find(q, 1, true) then
				found = found + 1
				local row = makeRow(ch, 0)
				row.Parent = treeScroll
				row.MouseButton1Click:Connect(function()
					local p = ch.Parent
					while p do expanded[p] = true p = p.Parent end
					exSearchBox.Text = ""
					selectInstance(ch)
				end)
			end
			walk(ch)
		end
	end
	walk(game)
	if found == 0 then
		new("TextLabel", { Size = UDim2.new(1, 0, 0, 30), BackgroundTransparency = 1,
			Text = "  (nenhum resultado)", TextColor3 = THEME.dim, TextSize = 14,
			TextXAlignment = Enum.TextXAlignment.Left, Parent = treeScroll })
	end
end

selectInstance = function(obj)
	selected = obj
	for o, row in pairs(rowRefs) do row.BackgroundTransparency = (o == obj) and 0 or 1 end
	renderProps()
end

local function getPropertyNames(obj)
	if type(getproperties) == "function" then
		local ok, res = pcall(getproperties, obj)
		if ok and type(res) == "table" then
			local names = {}
			for k, v in pairs(res) do
				if type(k) == "string" then names[#names + 1] = k
				elseif type(k) == "number" then names[#names + 1] = tostring(v) end
			end
			if type(gethiddenproperties) == "function" then
				local ok2, h = pcall(gethiddenproperties, obj)
				if ok2 and type(h) == "table" then
					for k in pairs(h) do if type(k) == "string" then names[#names + 1] = k end end
				end
			end
			if #names > 0 then return names end
		end
	end
	local names = {}
	for _, p in ipairs(COMMON_PROPS) do
		if pcall(function() return obj[p] end) then names[#names + 1] = p end
	end
	return names
end

			editBox.Size = UDim2.fromOffset(math.max(150, valBtn.AbsoluteSize.X), 22)
			editBox.Position = UDim2.fromOffset(valBtn.AbsolutePosition.X, valBtn.AbsolutePosition.Y)
			editBox.Text = editText(now)
			editBox.Visible = true
			task.defer(function() editBox:CaptureFocus() end)
		end
	end)
end

renderProps = function()
	clearList(propsScroll)
	if not selected then return end
	local obj = selected
	local head = new("Frame", { Size = UDim2.new(1, 0, 0, 32), BackgroundColor3 = THEME.header, BorderSizePixel = 0, Parent = propsScroll })
	new("TextLabel", { Size = UDim2.new(0, 26, 1, 0), Position = UDim2.new(0, 6, 0, 0), BackgroundTransparency = 1,
		Text = iconFor(obj), TextSize = 15, Parent = head })
	new("TextLabel", { Size = UDim2.new(1, -40, 1, 0), Position = UDim2.new(0, 34, 0, 0), BackgroundTransparency = 1,
		Text = obj.ClassName .. ' "' .. obj.Name .. '"', TextColor3 = THEME.text, TextSize = 14,
		Font = Enum.Font.SourceSansSemibold, TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd, Parent = head })
	local filter = propSearchBox.Text:lower()
	local names = getPropertyNames(obj)
	table.sort(names)
	for _, pname in ipairs(names) do
		if filter == "" or pname:lower():find(filter, 1, true) then makePropRow(obj, pname) end
	end
end

btnRefresh.MouseButton1Click:Connect(renderProps)

--[[ ---------- ATUALIZAÇÃO AUTOMÁTICA DA HIERARQUIA ---------- ]]
local updPending = false
local function scheduleUpdate()
	if updPending then return end
	updPending = true
	task.delay(0.5, function()
		updPending = false
		if selected and not pcall(function() return selected.Parent end) then selected = nil end
		if exSearchBox.Text == "" then renderTree() end
	end)
end
local function onChange(d)
	local p = d.Parent
	while p do
		if expanded[p] then scheduleUpdate() return end
		p = p.Parent
	end
end
game.DescendantAdded:Connect(onChange)
game.DescendantRemoving:Connect(onChange)

--[[ ---------- ARRASTAR PAINÉIS ---------- ]]
local function makeDraggable(frame, handle)
	local dragging, startP, startPos
	handle.InputBegan:Connect(function(inp)
		if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
			dragging = true; startP = inp.Position; startPos = frame.Position
		end
	end)
	handle.InputEnded:Connect(function(inp)
		if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then dragging = false end
	end)
	UIS.InputChanged:Connect(function(inp)
		if dragging and (inp.UserInputType == Enum.UserInputType.MouseMove or inp.UserInputType == Enum.UserInputType.Touch) then
			local d = inp.Position - startP
			frame.Position = startPos + UDim2.new(0, d.X, 0, d.Y)
		end
	end)
end
makeDraggable(explorer, exHeader)
makeDraggable(propsPanel, prHeader)
